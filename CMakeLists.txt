cmake_minimum_required(VERSION 3.25)
project(CoreSim VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()
include(CTest)
option(CORESIM_BUILD_APP "Build the GLFW application" ON)
option(CORESIM_FETCH_DEPENDENCIES "Fetch pinned dependencies instead of installed packages" ON)
option(CORESIM_WINDOW_TESTS "Enable tests requiring a display" OFF)
set(CORESIM_SANITIZER "none" CACHE STRING "none, address, or thread")
set_property(CACHE CORESIM_SANITIZER PROPERTY STRINGS none address thread)
include(cmake/ProjectOptions.cmake)
include(cmake/Dependencies.cmake)
add_library(coresim_core src/core/FrameTimer.cpp)
target_include_directories(coresim_core PUBLIC include)
coresim_target_options(coresim_core)
add_library(coresim_scene src/scene/Camera.cpp src/scene/Transform.cpp src/scene/DemoScene.cpp)
target_include_directories(coresim_scene PUBLIC include)
target_link_libraries(coresim_scene PUBLIC glm::glm)
coresim_target_options(coresim_scene)
if(CORESIM_BUILD_APP)
    add_library(coresim_glad STATIC third_party/glad/gl.c)
    target_include_directories(coresim_glad SYSTEM PUBLIC third_party/glad/include)
    add_library(coresim_platform src/core/Window.cpp)
    target_include_directories(coresim_platform PUBLIC include)
    target_link_libraries(coresim_platform PRIVATE glfw coresim_glad)
    target_compile_definitions(coresim_platform PRIVATE GLFW_INCLUDE_NONE)
    coresim_target_options(coresim_platform)

    add_library(coresim_renderer src/renderer/Buffers.cpp src/renderer/Shader.cpp
                                 src/renderer/Renderer.cpp)
    target_include_directories(coresim_renderer PUBLIC include)
    target_link_libraries(coresim_renderer PUBLIC coresim_scene PRIVATE coresim_glad)
    coresim_target_options(coresim_renderer)

    set(CORESIM_SHADER_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/shaders")
    file(MAKE_DIRECTORY "${CORESIM_SHADER_DIRECTORY}")
    configure_file(shaders/cube.vert shaders/cube.vert COPYONLY)
    configure_file(shaders/cube.frag shaders/cube.frag COPYONLY)
    add_executable(coresim src/main.cpp src/core/Application.cpp)
    target_link_libraries(coresim PRIVATE coresim_core coresim_platform coresim_renderer)
    target_compile_definitions(coresim PRIVATE CORESIM_SHADER_DIR="${CORESIM_SHADER_DIRECTORY}")
    coresim_target_options(coresim)
endif()
if(BUILD_TESTING)
    add_executable(coresim_tests tests/FrameTimerTests.cpp tests/SceneTests.cpp)
    target_link_libraries(coresim_tests PRIVATE coresim_core coresim_scene Catch2::Catch2WithMain)
    coresim_target_options(coresim_tests)
    add_test(NAME coresim.unit COMMAND coresim_tests)
    if(CORESIM_BUILD_APP)
        add_test(NAME coresim.help COMMAND coresim --help)
        add_test(NAME coresim.invalid_cli COMMAND coresim --frames 0)
        set_tests_properties(coresim.invalid_cli PROPERTIES WILL_FAIL TRUE)
        if(CORESIM_WINDOW_TESTS)
            add_executable(coresim_renderer_tests tests/RendererTests.cpp)
            target_link_libraries(coresim_renderer_tests PRIVATE coresim_platform coresim_renderer
                                  coresim_glad Catch2::Catch2WithMain)
            target_compile_definitions(coresim_renderer_tests PRIVATE
                CORESIM_SHADER_DIR="${CORESIM_SHADER_DIRECTORY}"
                CORESIM_TEST_SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/shaders")
            coresim_target_options(coresim_renderer_tests)
            add_test(NAME coresim.renderer COMMAND coresim_renderer_tests)
            set_tests_properties(coresim.renderer PROPERTIES TIMEOUT 30 LABELS display)
            add_test(NAME coresim.window_smoke COMMAND coresim --frames 5)
            set_tests_properties(coresim.window_smoke PROPERTIES TIMEOUT 20 LABELS display)
        endif()
    endif()
endif()
